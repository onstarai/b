 20: '자료 활용의 윤리성과 개인정보 보호의 중요성을 인식하고 책임감 있게 탐구 결과를 공유함. 분석의 정확성과 사회적 유용성을 함께 고려하며 데이터 과학을 통해 지속 가능한 가치를 창출하려는 '
     '성찰적인 태도가 돋보임.'}

class 데이터과학_과세특_조합기:
    def __init__(self, 창):
        self.창 = 창
        창.title("데이터 과학 과세특 문장 조합기")
        창.geometry("1080x760")
        창.minsize(760, 560)
        틀 = ttk.Frame(창, padding=18)
        틀.pack(fill="both", expand=True)
        ttk.Label(틀, text="데이터 과학 · 과세특 문장 조합기", font=("맑은 고딕", 19, "bold")).pack(anchor="w")
        ttk.Label(틀, text="시작 1개 + 서로 다른 본문 2개 + 마무리 1개 / 생성 후 문장 수정 가능").pack(anchor="w", pady=(6, 12))
        선택틀 = ttk.LabelFrame(틀, text="문장 선택", padding=12)
        선택틀.pack(fill="x")
        self.선택 = {}
        구성 = [("시작", 시작_멘트), ("본문 ①", 본문_멘트), ("본문 ②", 본문_멘트), ("마무리", 마무리_멘트)]
        for 행, (이름, 사전) in enumerate(구성):
            ttk.Label(선택틀, text=이름, width=10).grid(row=행, column=0, sticky="w", pady=6)
            콤보 = ttk.Combobox(선택틀, state="readonly", width=105,
                values=[f"{번호:02d}. {문장[:57]}…" for 번호, 문장 in 사전.items()])
            콤보.grid(row=행, column=1, sticky="ew", padx=8, pady=6)
            콤보.current([0, 0, 1, 0][행])
            self.선택[이름] = 콤보
        선택틀.columnconfigure(1, weight=1)
        단추틀 = ttk.Frame(틀)
        단추틀.pack(fill="x", pady=12)
        for 이름, 동작 in [("문장 생성", self.생성), ("결과 복사", self.복사), ("TXT 저장", self.저장), ("초기화", self.초기화)]:
            ttk.Button(단추틀, text=이름, command=동작).pack(side="left", padx=(0, 9))
        self.상태 = ttk.Label(틀, text="문장 생성 단추를 누르세요.")
        self.상태.pack(anchor="w", pady=(0, 7))
        결과틀 = ttk.LabelFrame(틀, text="생성된 과세특 (직접 수정 가능)", padding=9)
        결과틀.pack(fill="both", expand=True)
        self.결과 = tk.Text(결과틀, wrap="word", font=("맑은 고딕", 12), undo=True, spacing2=5)
        스크롤 = ttk.Scrollbar(결과틀, command=self.결과.yview)
        self.결과.configure(yscrollcommand=스크롤.set)
        self.결과.pack(side="left", fill="both", expand=True)
        스크롤.pack(side="right", fill="y")
        self.결과.bind("<KeyRelease>", self.글자수)

    def 번호(self, 이름):
        return self.선택[이름].current() + 1

    def 글자수(self, 사건=None):
        문장 = self.결과.get("1.0", "end-1c")
        self.상태.config(text=f"공백 포함 {len(문장)}자 / 공백 제외 {len(''.join(문장.split()))}자 / 목표 약 600자")

    def 생성(self):
        첫째, 둘째 = self.번호("본문 ①"), self.번호("본문 ②")
        if 첫째 == 둘째:
            messagebox.showwarning("본문 중복 선택", "본문 ①과 본문 ②는 서로 다른 번호를 선택해야 합니다.")
            return
        문장 = " ".join([시작_멘트[self.번호("시작")], 본문_멘트[첫째], 본문_멘트[둘째], 마무리_멘트[self.번호("마무리")]])
        self.결과.delete("1.0", "end")
        self.결과.insert("1.0", 문장)
        self.글자수()

    def 복사(self):
        문장 = self.결과.get("1.0", "end-1c").strip()
        if not 문장:
            messagebox.showinfo("안내", "먼저 문장을 생성하세요.")
            return
        self.창.clipboard_clear()
        self.창.clipboard_append(문장)
        messagebox.showinfo("복사 완료", "과세특 문장을 복사했습니다.")

    def 저장(self):
        문장 = self.결과.get("1.0", "end-1c").strip()
        if not 문장:
            messagebox.showinfo("안내", "먼저 문장을 생성하세요.")
            return
        경로 = filedialog.asksaveasfilename(defaultextension=".txt", filetypes=[("텍스트 파일", "*.txt")],
            initialfile=f"데이터과학_과세특_{datetime.now():%Y%m%d_%H%M%S}.txt")
        if 경로:
            with open(경로, "w", encoding="utf-8-sig") as 파일:
                파일.write(문장)
            messagebox.showinfo("저장 완료", "텍스트 파일로 저장했습니다.")

    def 초기화(self):
        self.결과.delete("1.0", "end")
        self.글자수()

if __name__ == "__main__":
    창 = tk.Tk()
    앱 = 데이터과학_과세특_조합기(창)
    창.mainloop()
