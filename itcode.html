class 과세특_조합기:
    def __init__(self, 창):
        self.창 = 창
        창.title("인공지능 기초 과세특 문장 조합기")
        창.geometry("1050x740")
        창.minsize(760, 550)
        바탕 = ttk.Frame(창, padding=18)
        바탕.pack(fill="both", expand=True)
        ttk.Label(바탕, text="인공지능 기초 · 과세특 600자 조합기", font=("맑은 고딕", 18, "bold")).pack(anchor="w", pady=(0, 8))
        ttk.Label(바탕, text="시작 1개 + 본문 2개(서로 다른 번호) + 마무리 1개를 선택하세요.").pack(anchor="w", pady=(0, 12))
        선택틀 = ttk.LabelFrame(바탕, text="문장 선택", padding=12)
        선택틀.pack(fill="x")
        self.선택값 = {}
        구성 = [("시작", 시작_멘트), ("본문 ①", 본문_멘트), ("본문 ②", 본문_멘트), ("마무리", 마무리_멘트)]
        for 행, (이름, 사전) in enumerate(구성):
            ttk.Label(선택틀, text=이름, width=10).grid(row=행, column=0, sticky="w", pady=5)
            값 = tk.StringVar()
            콤보 = ttk.Combobox(선택틀, textvariable=값, state="readonly", width=100,
                values=[f"{번호:02d}. {문장[:48]}..." for 번호, 문장 in 사전.items()])
            콤보.grid(row=행, column=1, sticky="ew", padx=6, pady=5)
            콤보.current(행 if 행 < 2 else (1 if 행 == 2 else 0))
            self.선택값[이름] = 값
        선택틀.columnconfigure(1, weight=1)
        단추틀 = ttk.Frame(바탕)
        단추틀.pack(fill="x", pady=12)
        ttk.Button(단추틀, text="문장 생성", command=self.생성).pack(side="left", padx=(0, 8))
        ttk.Button(단추틀, text="결과 복사", command=self.복사).pack(side="left", padx=(0, 8))
        ttk.Button(단추틀, text="TXT 저장", command=self.저장).pack(side="left", padx=(0, 8))
        ttk.Button(단추틀, text="초기화", command=self.초기화).pack(side="left")
        self.상태 = ttk.Label(바탕, text="문장 생성 단추를 누르세요.")
        self.상태.pack(anchor="w", pady=(0, 6))
        결과틀 = ttk.LabelFrame(바탕, text="과세특 결과 (수정 가능)", padding=8)
        결과틀.pack(fill="both", expand=True)
        self.결과 = tk.Text(결과틀, wrap="word", font=("맑은 고딕", 12), undo=True, spacing2=5)
        막대 = ttk.Scrollbar(결과틀, orient="vertical", command=self.결과.yview)
        self.결과.configure(yscrollcommand=막대.set)
        self.결과.pack(side="left", fill="both", expand=True)
        막대.pack(side="right", fill="y")

    def 번호(self, 이름):
        return int(self.선택값[이름].get().split(".", 1)[0])

    def 생성(self):
        첫째, 둘째 = self.번호("본문 ①"), self.번호("본문 ②")
        if 첫째 == 둘째:
            messagebox.showwarning("중복 선택", "본문 멘트 ①과 ②는 서로 다른 번호를 선택해야 합니다.")
            return
        문장들 = [시작_멘트[self.번호("시작")], 본문_멘트[첫째],
                 본문_멘트[둘째], 마무리_멘트[self.번호("마무리")]]
        전체 = " ".join(문장들)
        self.결과.delete("1.0", "end")
        self.결과.insert("1.0", 전체)
        self.상태.config(text=f"생성 완료 · 공백 포함 {len(전체)}자 / 공백 제외 {len(''.join(전체.split()))}자")

    def 복사(self):
        내용 = self.결과.get("1.0", "end-1c").strip()
        if not 내용:
            messagebox.showinfo("안내", "먼저 문장을 생성하세요.")
            return
        self.창.clipboard_clear()
        self.창.clipboard_append(내용)
        messagebox.showinfo("복사 완료", "결과를 클립보드에 복사했습니다.")

    def 저장(self):
        내용 = self.결과.get("1.0", "end-1c").strip()
        if not 내용:
            messagebox.showinfo("안내", "먼저 문장을 생성하세요.")
            return
        경로 = filedialog.asksaveasfilename(defaultextension=".txt", filetypes=[("텍스트", "*.txt")],
            initialfile=f"인공지능기초_과세특_{datetime.now():%Y%m%d_%H%M%S}.txt")
        if 경로:
            with open(경로, "w", encoding="utf-8-sig") as 파일:
                파일.write(내용)
            messagebox.showinfo("저장 완료", f"저장했습니다.\n{경로}")

    def 초기화(self):
        self.결과.delete("1.0", "end")
        self.상태.config(text="초기화되었습니다.")
