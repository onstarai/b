        가, 나, 다, 라 = [변수.get() for 변수 in self.선택값]
        문장 = " ".join([시작_멘트[가], 본문_멘트[나], 본문_멘트[다], 마무리_멘트[라]])

    def 복사(self):

        if not 내용.strip():

            return
        self.창.clipboard_clear()
        self.창.clipboard_append(내용)
        self.창.update()
        messagebox.showinfo("복사 완료", "과세특 문장을 복사했습니다.")

    def 저장(self):
        내용 = self.출력.get("1.0", "end-1c")
        if not 내용.strip():
            messagebox.showinfo("안내", "저장할 문장이 없습니다.")
            return
        경로 = filedialog.asksaveasfilename(defaultextension=".txt", initialfile="정보과학_과세특.txt",
                                          filetypes=[("텍스트 파일", "*.txt")])
        if 경로:
            with open(경로, "w", encoding="utf-8-sig") as 파일:

            messagebox.showinfo("저장 완료", "텍스트 파일을 저장했습니다.")
